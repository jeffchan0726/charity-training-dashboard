/**
 * Google Apps Script Backend for Jeff 減脂 Recomp 挑戰 (Charity Workout Log)
 *
 * IMPORTANT FIXES & IMPROVEMENTS:
 * - PIN Leading Zero Fix: PIN is ALWAYS stored and compared as STRING + explicit Text format in sheet.
 *   This prevents Google Sheets from auto-converting "0123" → 123.
 * - Clear distinction between "user not exist" and "wrong PIN".
 * - Per-user logs + per-user Workout Sets (3 defaults auto-seeded on registration).
 * - FULL CRUD for Workout Logs (2026 update + major UX improvements):
 *   - addLog (single set, with session_id support)
 *   - addWorkout (bulk whole workout/session in one call) — recommended for new finished workouts
 *   - updateLog (HIGHLY IMPROVED):
 *       • Recommended: by unique "id" (fast, simple, reliable for single set quick edits)
 *       • Alternative: by "session_id + exercise + setIndex" (0-based) for session-based edits
 *       • Auto-recalculates volume (weight×reps, body_weight×reps, or treadmill km) when not provided
 *       • Very clear error messages for frontend (setIndex out of range, no matching row, etc.)
 *   - replaceSession (NEW — powerful for History "big edit"):
 *       • Atomically replace an entire past workout/session (delete old rows + insert new)
 *       • Frontend can send the full updated workout shape with same session_id
 *   - deleteSession (IMPROVED): returns deleted count, clearer messages for legacy data
 *   - deleteLog (single set by id)
 *   - getLogs returns flat rows (with session_id) — frontend groups by session_id/date
 *     Optional ?recentDays=90 for faster initial load (filters by date >= cutoff)
 *   - getYugongLeaderboard (GET or POST): all players total moved tonnes (weight volume kg ÷ 1000)
 * - Sheet schema: ["id", "session_id", "date", "exercise", "reps", "weight", "body_weight", "duration", "incline", "speed", "notes", "volume"]
 *   • duration: seconds (time+reps holds) OR minutes (跑步機 treadmill)
 *   • body_weight: 自重動作體重 (kg)。incline / speed: 跑步機。volume = kg 或 km。
 * - Automatic column migration for old sheets (inserts session_id column safely)
 * - All write ops return consistent { status: "success" | "error", message?, ...extra info }
 * - Excellent legacy data (no session_id) compatibility
 *
 * RECOMMENDED FRONTEND PATTERNS (for mobile History editing):
 * - Quick single-set edit (weight/reps/notes): use updateLog + { id: "..." }
 * - Major rewrite of a whole past training day: use replaceSession (or deleteSession + addWorkout)
 * - Delete whole training: deleteSession (now returns deleted count)
 *
 * DEPLOYMENT:
 * 1. Create a new Google Sheet.
 * 2. Paste this entire script into Extensions > Apps Script.
 * 3. Deploy > New deployment > Web app:
 *    - Execute as: Me
 *    - Who has access: Anyone, even anonymous
 * 4. Copy the /exec URL into your index.html (search for APPS_SCRIPT_URL).
 * 5. After any change, create a NEW deployment version (do not just save).
 *
 * MIGRATION NOTE for existing users:
 * After deploying this version, the first write (add/update/replace) for an old user will
 * automatically insert a "session_id" column (old rows get blank session_id, which is fine —
 * frontend has fallback grouping). New replaceSession / improved updateLog handle legacy gracefully.
 */
function doGet(e) {
  if (!e || !e.parameter || !e.parameter.action) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error", message: "No action provided"
    })).setMimeType(ContentService.MimeType.JSON);
  }

  const action = e.parameter.action;
  const user = e.parameter.user;

  if (action === "getLogs" && user) {
    const auth = _authorizeUser(user, e.parameter.pin);
    if (auth.error) return auth.error;
    const recentDays = parseInt(e.parameter.recentDays, 10);
    return getUserLogs(auth.user, isNaN(recentDays) || recentDays <= 0 ? null : recentDays);
  }
  if (action === "getWorkoutSets" && user) {
    const auth = _authorizeUser(user, e.parameter.pin);
    if (auth.error) return auth.error;
    return getUserWorkoutSets(auth.user);
  }
  if (action === "getYugongLeaderboard") {
    return getYugongLeaderboard();
  }

  return ContentService.createTextOutput(JSON.stringify({
    status: "error", message: "Invalid action or missing user"
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const action = data.action;

    if (action === "addUser") {
      return addNewUser(data.user, data.pin);
    }
    if (action === "login") {
      return verifyLogin(data.user, data.pin);
    }
    if (action === "addLog") {
      const auth = _authorizeUser(data.user, data.pin);
      if (auth.error) return auth.error;
      return addUserLog(auth.user, data.log);
    }
    if (action === "addWorkout") {
      const auth = _authorizeUser(data.user, data.pin);
      if (auth.error) return auth.error;
      return addUserWorkout(auth.user, data.workout);
    }
    if (action === "updateLog") {
      const auth = _authorizeUser(data.user, data.pin);
      if (auth.error) return auth.error;
      return updateUserLog(auth.user, data.log);
    }
    if (action === "deleteLog") {
      const auth = _authorizeUser(data.user, data.pin);
      if (auth.error) return auth.error;
      return deleteUserLog(auth.user, data.logId);
    }
    if (action === "deleteSession") {
      const auth = _authorizeUser(data.user, data.pin);
      if (auth.error) return auth.error;
      return deleteUserSession(auth.user, data.sessionId);
    }
    if (action === "replaceSession" || action === "updateSession") {
      const auth = _authorizeUser(data.user, data.pin);
      if (auth.error) return auth.error;
      return replaceUserSession(auth.user, data.sessionId, data.workout);
    }
    if (action === "saveWorkoutSet") {
      const auth = _authorizeUser(data.user, data.pin);
      if (auth.error) return auth.error;
      return saveUserWorkoutSet(auth.user, data.set);
    }
    if (action === "deleteWorkoutSet") {
      const auth = _authorizeUser(data.user, data.pin);
      if (auth.error) return auth.error;
      return deleteUserWorkoutSet(auth.user, data.setId);
    }
    if (action === "getYugongLeaderboard") {
      return getYugongLeaderboard();
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: "error", message: "Unknown action"
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error", message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function _normalizeUser(u) {
  return String(u || '').trim().toLowerCase().replace(/[^a-z0-9]/g, '');
}

function _normalizePin(p) {
  return String(p || '').trim().replace(/\D/g, '');
}

function _authError(message) {
  return ContentService.createTextOutput(JSON.stringify({
    status: "error", message: message || "未授權"
  })).setMimeType(ContentService.MimeType.JSON);
}

function _pinMatches(username, pin) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const userSheet = ss.getSheetByName("Users");
  if (!userSheet) return false;
  const cleanUser = _normalizeUser(username);
  const cleanPin = _normalizePin(pin);
  if (!cleanUser || !cleanPin) return false;
  const data = userSheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    const sheetUsername = _normalizeUser(data[i][0]);
    const sheetPin = _normalizePin(data[i][1]);
    if (sheetUsername === cleanUser) return sheetPin === cleanPin;
  }
  return false;
}

function _authorizeUser(username, pin) {
  const cleanUser = _normalizeUser(username);
  const cleanPin = _normalizePin(pin);
  if (!cleanUser) return { error: _authError("缺少用戶") };
  if (!cleanPin) return { error: _authError("請重新登入（缺少 PIN）") };
  if (!_pinMatches(cleanUser, cleanPin)) return { error: _authError("PIN 碼錯誤或未授權") };
  return { user: cleanUser };
}

// ==================== USER MANAGEMENT (PIN LEADING ZERO SAFE) ====================

/**
 * Register new user.
 * PIN is normalized to digits-only string on both sides.
 * CRITICAL: Write PIN using explicit setValue(String(pin)) + setNumberFormat('@')
 * so that leading zeros (e.g. 0123) are preserved in the Google Sheet.
 */
function addNewUser(username, pin) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const userSheet = ss.getSheetByName("Users") || ss.insertSheet("Users");

  // Normalize incoming
  const cleanUser = String(username || '').trim().toLowerCase().replace(/[^a-z0-9]/g, '');
  const cleanPin  = String(pin || '').trim().replace(/\D/g, '');

  if (!cleanUser || !cleanPin) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error", message: "用戶名或 PIN 格式不正確"
    })).setMimeType(ContentService.MimeType.JSON);
  }

  // Check duplicate username (normalized)
  const data = userSheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    const existing = String(data[i][0]).trim().toLowerCase().replace(/[^a-z0-9]/g, '');
    if (existing === cleanUser) {
      return ContentService.createTextOutput(JSON.stringify({
        status: "error", message: "用戶名已被使用"
      })).setMimeType(ContentService.MimeType.JSON);
    }
  }

  // Create per-user log sheet (if not exists) — new schema with session_id for full workout support
  _ensureUserLogSheet(ss, cleanUser);

  // Write header if first time
  if (userSheet.getLastRow() === 0) {
    userSheet.appendRow(["username", "pin", "created"]);
    // Force the pin column header cell to text format (good practice)
    userSheet.getRange(1, 2).setNumberFormat('@');
  }

  // === PIN LEADING ZERO FIX: Write as explicit string + text format ===
  const newRow = userSheet.getLastRow() + 1;
  userSheet.getRange(newRow, 1).setValue(cleanUser);
  
  const pinCell = userSheet.getRange(newRow, 2);
  pinCell.setValue(cleanPin);           // IMPORTANT: pass as string "0123"
  pinCell.setNumberFormat('@');         // Force TEXT format so leading zero is never dropped

  userSheet.getRange(newRow, 3).setValue(new Date());

  // NOTE (cleaned): No longer seed any default Workout Sets on registration.
  // Workout_Sets sheet is now strictly for user-created custom Sets only.
  // The 3 fixed Training Days live exclusively in frontend TRAINING_DAYS.

  return ContentService.createTextOutput(JSON.stringify({
    status: "success"
  })).setMimeType(ContentService.MimeType.JSON);
}

/**
 * Verify login with clear error messages.
 * Both username and PIN are normalized to string before comparison.
 */
function verifyLogin(username, pin) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const userSheet = ss.getSheetByName("Users");
  
  if (!userSheet) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error", 
      message: "用戶不存在，請確認用戶名或先註冊"
    })).setMimeType(ContentService.MimeType.JSON);
  }

  const cleanUser = String(username || '').trim().toLowerCase().replace(/[^a-z0-9]/g, '');
  const cleanPin  = String(pin || '').trim().replace(/\D/g, '');

  const data = userSheet.getDataRange().getValues();
  
  for (let i = 1; i < data.length; i++) {
    const sheetUsername = String(data[i][0]).trim().toLowerCase().replace(/[^a-z0-9]/g, '');
    // Read whatever is in the cell as string (the setNumberFormat('@') guarantees the original digits)
    const sheetPin = String(data[i][1]).trim().replace(/\D/g, '');

    if (sheetUsername === cleanUser) {
      if (sheetPin === cleanPin) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "success"
        })).setMimeType(ContentService.MimeType.JSON);
      } else {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", 
          message: "PIN 碼錯誤，請重新輸入"
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }
  }

  return ContentService.createTextOutput(JSON.stringify({
    status: "error", 
    message: "用戶不存在，請確認用戶名或先註冊"
  })).setMimeType(ContentService.MimeType.JSON);
}

// ==================== WORKOUT LOGS (FULL CRUD + session_id support) ====================

/**
 * Centralized helper: ensure the per-user Workout_Log sheet exists and has the correct schema.
 * Automatically migrates old sheets by adding "session_id" column if missing.
 * Safe to call multiple times.
 */
function _ensureUserLogSheet(ss, username) {
  const sheetName = "Workout_Log_" + username;
  let sheet = ss.getSheetByName(sheetName);

  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(["id", "session_id", "date", "exercise", "reps", "weight", "body_weight", "duration", "incline", "speed", "notes", "volume", "session_notes"]);
    return sheet;
  }

  // Auto-migrate old sheet: insert session_id / duration / incline / speed columns if missing
  let headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
  if (!headers.includes("session_id")) {
    sheet.insertColumnAfter(1);
    sheet.getRange(1, 2).setValue("session_id");
    headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
  }
  _insertColumnIfMissing(sheet, headers, "body_weight", "weight");
  headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
  _insertColumnIfMissing(sheet, headers, "duration", "body_weight");
  headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
  _insertColumnIfMissing(sheet, headers, "incline", "duration");
  headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
  _insertColumnIfMissing(sheet, headers, "speed", "incline");
  headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
  _insertColumnIfMissing(sheet, headers, "session_notes", "notes");

  return sheet;
}

function _insertColumnIfMissing(sheet, headers, columnName, afterColumnName) {
  if (headers.includes(columnName)) return;
  const afterIdx = headers.indexOf(afterColumnName);
  if (afterIdx >= 0) {
    sheet.insertColumnAfter(afterIdx + 1);
    sheet.getRange(1, afterIdx + 2).setValue(columnName);
  } else {
    const lastCol = sheet.getLastColumn();
    sheet.insertColumnAfter(lastCol);
    sheet.getRange(1, lastCol + 1).setValue(columnName);
  }
}

/** Treadmill distance (km) = speed (km/h) × time (hours) */
function _calculateTreadmillDistanceKm(speed, durationMin) {
  const sp = parseFloat(speed) || 0;
  const mins = parseInt(durationMin) || 0;
  if (sp <= 0 || mins <= 0) return 0;
  return Math.round(sp * (mins / 60) * 100) / 100;
}

/** volume = kg (weights/bodyweight×reps) OR km distance (treadmill) */
function _calculateVolume(weight, reps, duration, incline, speed, bodyWeight) {
  const sp = parseFloat(speed) || 0;
  if (sp > 0) return _calculateTreadmillDistanceKm(sp, duration);
  const d = parseInt(duration) || 0;
  const r = parseInt(reps) || 0;
  const bw = parseFloat(bodyWeight) || 0;
  if (bw > 0 && r > 0) return Math.round(bw * r);
  if (d > 0 && r > 0) return 0;
  const w = parseFloat(weight) || 0;
  return Math.round(w * r);
}

/**
 * Helper: safely append a log row matching the current sheet headers.
 * Supports both old sheets (no session_id) and new schema.
 */
function _appendLogRow(sheet, log) {
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
  const row = [];

  for (let h of headers) {
    if (h === "id") {
      row.push(log.id != null && String(log.id).trim() !== "" ? log.id : (Date.now() + Math.random()));
    } else if (h === "session_id") {
      row.push(log.session_id || log.sessionId || '');
    } else if (h === "date") {
      row.push(log.date || '');
    } else if (h === "exercise") {
      row.push(log.exercise || '');
    } else if (h === "reps") {
      row.push(log.reps != null ? parseInt(log.reps) : 0);
    } else if (h === "weight") {
      row.push(log.weight != null ? parseFloat(log.weight) : 0);
    } else if (h === "body_weight") {
      row.push(log.body_weight != null ? parseFloat(log.body_weight) : 0);
    } else if (h === "duration") {
      row.push(log.duration != null ? parseInt(log.duration) : 0);
    } else if (h === "incline") {
      row.push(log.incline != null ? parseFloat(log.incline) : 0);
    } else if (h === "speed") {
      row.push(log.speed != null ? parseFloat(log.speed) : 0);
    } else if (h === "notes") {
      row.push(log.notes || '');
    } else if (h === "session_notes") {
      row.push(log.session_notes || log.sessionNotes || '');
    } else if (h === "volume") {
      // Prefer explicit volume, otherwise auto-calc
      const vol = (log.volume != null) ? parseFloat(log.volume) : _calculateVolume(log.weight, log.reps, log.duration, log.incline, log.speed, log.body_weight);
      row.push(vol);
    } else {
      // legacy columns like old "sets" or unknown — put empty or value if present
      row.push(log[h] != null ? log[h] : '');
    }
  }
  sheet.appendRow(row);
}

// ==================== 愚公移山排行榜（全玩家累積搬山噸數） ====================

/**
 * Sum weight-based training volume (kg) for one user's log sheet.
 * Treadmill rows (speed > 0) are excluded — their volume is km, not kg.
 */
function _sumUserWeightVolumeFromSheet(sheet) {
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return 0;

  const headers = data[0].map(function(h) { return String(h).trim(); });
  const idx = {};
  headers.forEach(function(h, i) { idx[h] = i; });

  let totalKg = 0;
  for (let r = 1; r < data.length; r++) {
    const row = data[r];
    const speed = idx.speed != null ? (parseFloat(row[idx.speed]) || 0) : 0;
    if (speed > 0) continue;

    let vol = idx.volume != null ? (parseFloat(row[idx.volume]) || 0) : 0;
    if (!vol || vol <= 0) {
      vol = _calculateVolume(
        idx.weight != null ? row[idx.weight] : 0,
        idx.reps != null ? row[idx.reps] : 0,
        idx.duration != null ? row[idx.duration] : 0,
        idx.incline != null ? row[idx.incline] : 0,
        speed,
        idx.body_weight != null ? row[idx.body_weight] : 0
      );
    }
    totalKg += vol;
  }
  return Math.round(totalKg);
}

/**
 * Public leaderboard: every registered player's lifetime moved tonnes.
 * Sorted high → low. No PIN required (read-only aggregate).
 */
function getYugongLeaderboard() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const userSheet = ss.getSheetByName("Users");
    const seen = {};
    const entries = [];

    if (userSheet && userSheet.getLastRow() >= 2) {
      const rows = userSheet.getDataRange().getValues().slice(1);
      rows.forEach(function(row) {
        const username = String(row[0] || '').trim().toLowerCase().replace(/[^a-z0-9]/g, '');
        if (!username || seen[username]) return;
        seen[username] = true;
        const logSheet = ss.getSheetByName("Workout_Log_" + username);
        const weightKg = logSheet ? _sumUserWeightVolumeFromSheet(logSheet) : 0;
        entries.push({
          user: username,
          weightKg: weightKg,
          tonnes: Math.round(weightKg / 10) / 100
        });
      });
    }

    // Fallback: include any Workout_Log_* sheet not in Users (legacy / manual sheets)
    ss.getSheets().forEach(function(sheet) {
      const name = sheet.getName();
      if (name.indexOf("Workout_Log_") !== 0) return;
      const username = name.substring("Workout_Log_".length).trim().toLowerCase().replace(/[^a-z0-9]/g, '');
      if (!username || seen[username]) return;
      seen[username] = true;
      const weightKg = _sumUserWeightVolumeFromSheet(sheet);
      entries.push({
        user: username,
        weightKg: weightKg,
        tonnes: Math.round(weightKg / 10) / 100
      });
    });

    entries.sort(function(a, b) { return b.weightKg - a.weightKg; });

    const totalWeightKg = entries.reduce(function(sum, entry) { return sum + (entry.weightKg || 0); }, 0);
    const totalTonnes = Math.round(totalWeightKg / 10) / 100;

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      leaderboard: entries,
      totalWeightKg: totalWeightKg,
      totalTonnes: totalTonnes,
      updatedAt: new Date().toISOString()
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * Returns all log rows for the user as flat array.
 * Each row now includes "session_id" (when column exists).
 * Frontend is responsible for grouping by session_id or date (see loadUserLogs in index.html).
 * This flat format is most flexible for CRUD.
 */
function getUserLogs(username, recentDays) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Workout_Log_" + username);
  if (!sheet) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);

  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);

  const headers = data[0].map(h => String(h).trim());
  let cutoffStr = null;
  if (recentDays && recentDays > 0) {
    const cutoff = new Date();
    cutoff.setDate(cutoff.getDate() - recentDays);
    cutoffStr = Utilities.formatDate(cutoff, Session.getScriptTimeZone(), "yyyy-MM-dd");
  }

  const logs = data.slice(1).map(row => {
    let obj = {};
    headers.forEach((h, i) => {
      let val = row[i];
      if (h === "date" && val) {
        if (val instanceof Date) {
          // 強制轉為乾淨 YYYY-MM-DD，使用 script 時區
          val = Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd");
        } else {
          // 如果是字串或 ISO，強制取本地日期部分
          val = String(val).split('T')[0];
        }
      }
      obj[h] = val;
    });
    return obj;
  }).filter(function(obj) {
    if (!cutoffStr) return true;
    const d = String(obj.date || '').split('T')[0];
    return d >= cutoffStr;
  });

  return ContentService.createTextOutput(JSON.stringify(logs)).setMimeType(ContentService.MimeType.JSON);
}

/**
 * Add single set log.
 * Frontend typically calls this in a loop for each set in a workout (passing session_id).
 */
function addUserLog(username, log) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = _ensureUserLogSheet(ss, username);
  _appendLogRow(sheet, log || {});
  return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
}

/**
 * Bulk add a full workout / session in one call.
 * Recommended for finishing a complete training session.
 * Frontend example: { action: "addWorkout", user, workout: { session_id, date, exercises: [{name, sets: [...] }], notes? } }
 */
function addUserWorkout(username, workout) {
  try {
    if (!username || !workout || !Array.isArray(workout.exercises)) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Invalid workout data" })).setMimeType(ContentService.MimeType.JSON);
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = _ensureUserLogSheet(ss, username);

    const sessionId = String(workout.session_id || workout.id || Date.now());
    const dateStr = workout.date || '';
    const sessionNotes = workout.notes || workout.session_notes || '';
    let added = 0;

    workout.exercises.forEach(ex => {
      (ex.sets || []).forEach(set => {
        const log = {
          id: set.id || set._clientLogId || (Date.now() + Math.random()),
          session_id: sessionId,
          date: dateStr,
          exercise: ex.name || '',
          reps: set.reps || 0,
          weight: set.weight || 0,
          body_weight: set.body_weight != null ? parseFloat(set.body_weight) : 0,
          duration: set.duration != null ? parseInt(set.duration) : 0,
          incline: set.incline != null ? parseFloat(set.incline) : 0,
          speed: set.speed != null ? parseFloat(set.speed) : 0,
          notes: set.notes || '',
          session_notes: sessionNotes,
          volume: set.volume
        };
        _appendLogRow(sheet, log);
        added++;
      });
    });

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      session_id: sessionId,
      added: added
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * ========================================================================
 * MAJORLY IMPROVED: updateUserLog
 * ========================================================================
 *
 * Much more intuitive and robust for frontend (especially mobile History editing).
 *
 * Two supported update methods (in priority order):
 *
 * 1. By unique ID (HIGHLY RECOMMENDED for "quick single set edit")
 *    { action: "updateLog", user: "...", log: { id: "the-log-id", weight: 85, reps: 5, notes: "..." } }
 *    - Fast, reliable, no need to know exercise name or setIndex.
 *    - Frontend can store the "id" of each set when rendering history.
 *
 * 2. By session + exercise + setIndex (0-based) — kept for flexibility
 *    { action: "updateLog", user: "...", log: { session_id: "...", exercise: "硬拉 (Deadlift)", setIndex: 0, weight: 82.5, reps: 6 } }
 *    - Useful when you only have the logical position inside a session.
 *    - Validation is now strict and gives excellent error messages.
 *
 * Additional improvements:
 * - If "volume" is not provided (or null), backend **automatically recalculates** it as weight × reps.
 * - Partial updates supported (you only need to send the fields you want to change).
 * - Excellent error messages (in Chinese where appropriate for HK users).
 * - Legacy rows (missing session_id) are fully supported when using the `id` method.
 * - Always ensures sheet schema.
 */
function updateUserLog(username, log) {
  try {
    if (!username || !log) {
      return ContentService.createTextOutput(JSON.stringify({ 
        status: "error", 
        message: "缺少用戶或更新資料 (Missing user or log data)" 
      })).setMimeType(ContentService.MimeType.JSON);
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = _ensureUserLogSheet(ss, username);

    const data = sheet.getDataRange().getValues();
    if (data.length < 2) {
      return ContentService.createTextOutput(JSON.stringify({ 
        status: "error", 
        message: "此用戶尚無訓練記錄" 
      })).setMimeType(ContentService.MimeType.JSON);
    }

    const headers = data[0].map(h => String(h).trim());
    const idCol = headers.indexOf("id");
    const sidCol = headers.indexOf("session_id");
    const exCol = headers.indexOf("exercise");
    const repsCol = headers.indexOf("reps");
    const weightCol = headers.indexOf("weight");
    const notesCol = headers.indexOf("notes");
    const volCol = headers.indexOf("volume");
    const dateCol = headers.indexOf("date");

    let targetRowIndex = -1; // 1-based sheet row number
    let matchedBy = "";

    // ==================== METHOD 1: Direct by unique id (RECOMMENDED) ====================
    if (log.id != null && String(log.id).trim() !== "") {
      const searchCol = idCol >= 0 ? idCol : 0;
      const targetId = String(log.id).trim();

      for (let i = 1; i < data.length; i++) {
        const rowId = String(data[i][searchCol] || "").trim();
        if (rowId === targetId) {
          targetRowIndex = i + 1;
          matchedBy = "id";
          break;
        }
      }

      if (targetRowIndex < 0) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error",
          message: `找不到指定的記錄 (Log with id "${log.id}" not found)`
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ==================== METHOD 2: session_id + exercise + setIndex (0-based) ====================
    if (targetRowIndex < 0 && log.session_id != null && log.exercise != null && log.setIndex != null) {
      const sessionIdStr = String(log.session_id).trim();
      const exName = String(log.exercise).trim();
      const setIdx = parseInt(log.setIndex, 10);

      if (sidCol < 0 || exCol < 0) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error",
          message: "工作表缺少 session_id 或 exercise 欄位，無法使用 session-based 更新"
        })).setMimeType(ContentService.MimeType.JSON);
      }

      // Collect all rows belonging to this session + this exact exercise name
      let matchingRows = [];
      for (let i = 1; i < data.length; i++) {
        const rowSid = String(data[i][sidCol] || "").trim();
        const rowEx = String(data[i][exCol] || "").trim();
        if (rowSid === sessionIdStr && rowEx === exName) {
          matchingRows.push({ sheetRow: i + 1, dataIdx: i });
        }
      }

      if (matchingRows.length === 0) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error",
          message: `在 session "${sessionIdStr}" 中找不到動作 "${exName}" 的任何組數`
        })).setMimeType(ContentService.MimeType.JSON);
      }

      if (setIdx < 0 || setIdx >= matchingRows.length) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error",
          message: `setIndex ${setIdx} 超出範圍。動作 "${exName}" 在此訓練 (session) 只有 ${matchingRows.length} 組（setIndex 必須是 0 ~ ${matchingRows.length - 1}）`
        })).setMimeType(ContentService.MimeType.JSON);
      }

      targetRowIndex = matchingRows[setIdx].sheetRow;
      matchedBy = "session+exercise+setIndex";
    }

    if (targetRowIndex < 0) {
      return ContentService.createTextOutput(JSON.stringify({
        status: "error",
        message: "無法定位要更新的記錄。請提供 log.id（推薦），或同時提供 session_id + exercise + setIndex"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ==================== PERFORM UPDATE (header-aware + auto volume) ====================
    // Collect what the caller wants to change
    const durationCol = headers.indexOf("duration");

    const incoming = {
      session_id: log.session_id != null ? log.session_id : log.sessionId,
      date: log.date,
      exercise: log.exercise,
      reps: log.reps,
      weight: log.weight,
      body_weight: log.body_weight,
      duration: log.duration,
      incline: log.incline,
      speed: log.speed,
      notes: log.notes,
      volume: log.volume
    };

    const inclineCol = headers.indexOf("incline");
    const speedCol = headers.indexOf("speed");
    const bodyWeightCol = headers.indexOf("body_weight");

    // Read current values for auto-volume calculation
    const currentRow = data[targetRowIndex - 1]; // 0-based data array
    const currentWeight = incoming.weight != null ? parseFloat(incoming.weight) : (weightCol >= 0 ? parseFloat(currentRow[weightCol]) || 0 : 0);
    const currentBodyWeight = incoming.body_weight != null ? parseFloat(incoming.body_weight) : (bodyWeightCol >= 0 ? parseFloat(currentRow[bodyWeightCol]) || 0 : 0);
    const currentReps = incoming.reps != null ? parseInt(incoming.reps) : (repsCol >= 0 ? parseInt(currentRow[repsCol]) || 0 : 0);
    const currentDuration = incoming.duration != null ? parseInt(incoming.duration) : (durationCol >= 0 ? parseInt(currentRow[durationCol]) || 0 : 0);
    const currentIncline = incoming.incline != null ? parseFloat(incoming.incline) : (inclineCol >= 0 ? parseFloat(currentRow[inclineCol]) || 0 : 0);
    const currentSpeed = incoming.speed != null ? parseFloat(incoming.speed) : (speedCol >= 0 ? parseFloat(currentRow[speedCol]) || 0 : 0);

    // Final values to write
    const finalUpdates = {
      "session_id": incoming.session_id,
      "date": incoming.date,
      "exercise": incoming.exercise,
      "reps": incoming.reps,
      "weight": incoming.weight,
      "body_weight": incoming.body_weight,
      "duration": incoming.duration,
      "incline": incoming.incline,
      "speed": incoming.speed,
      "notes": incoming.notes,
      "volume": (incoming.volume != null) ? parseFloat(incoming.volume) : _calculateVolume(currentWeight, currentReps, currentDuration, currentIncline, currentSpeed, currentBodyWeight)
    };

    let updatedFields = [];
    for (const [colName, val] of Object.entries(finalUpdates)) {
      if (val != null) {   // only write fields that were explicitly provided or auto-calculated
        const colIdx = headers.indexOf(colName);
        if (colIdx >= 0) {
          sheet.getRange(targetRowIndex, colIdx + 1).setValue(val);
          updatedFields.push(colName);
        }
      }
    }

    // Also update volume explicitly if we auto-calculated it and the column exists
    if (volCol >= 0 && incoming.volume == null) {
      // Already handled above because we always put the calculated value in finalUpdates["volume"]
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      message: "更新成功",
      matchedBy: matchedBy,
      updatedFields: updatedFields,
      row: targetRowIndex
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ 
      status: "error", 
      message: "更新失敗: " + err.toString() 
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * Delete a single log row by its id (one set).
 * Returns deleted count for consistency.
 */
function deleteUserLog(username, logId) {
  try {
    if (!username || logId == null) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Missing user or logId" })).setMimeType(ContentService.MimeType.JSON);
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Workout_Log_" + username);
    if (!sheet) return ContentService.createTextOutput(JSON.stringify({ status: "success", deleted: 0 })).setMimeType(ContentService.MimeType.JSON);

    const data = sheet.getDataRange().getValues();
    let deleted = 0;

    // Delete from bottom to avoid row index shifting
    for (let i = data.length - 1; i >= 1; i--) {
      if (String(data[i][0]).trim() === String(logId).trim()) {
        sheet.deleteRow(i + 1);
        deleted++;
        break;
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ 
      status: "success", 
      deleted: deleted 
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * ========================================================================
 * IMPROVED: deleteUserSession
 * ========================================================================
 *
 * Delete ALL logs belonging to a session_id (entire workout / one training day session).
 * Now returns the actual number of rows deleted (very useful for frontend confirmation).
 *
 * Robust against:
 * - Missing session_id column (legacy data)
 * - Empty / falsy sessionId
 * - Non-existent sessions
 *
 * Frontend example: { action: "deleteSession", user, sessionId: "..." }
 * Response on success: { status: "success", deleted: 12 }
 */
function deleteUserSession(username, sessionId) {
  try {
    if (!username || sessionId == null || String(sessionId).trim() === "") {
      return ContentService.createTextOutput(JSON.stringify({ 
        status: "error", 
        message: "缺少用戶或 sessionId (Missing user or sessionId)" 
      })).setMimeType(ContentService.MimeType.JSON);
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Workout_Log_" + username);
    if (!sheet) {
      return ContentService.createTextOutput(JSON.stringify({ 
        status: "success", 
        deleted: 0,
        message: "用戶記錄表不存在"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    const data = sheet.getDataRange().getValues();
    const headers = data[0].map(h => String(h).trim());
    const sidCol = headers.indexOf("session_id");
    const checkCol = sidCol >= 0 ? sidCol : -1;

    if (checkCol < 0) {
      // No session_id column yet — old data cannot be deleted by session
      return ContentService.createTextOutput(JSON.stringify({ 
        status: "success", 
        deleted: 0,
        message: "此工作表尚未有 session_id 欄位，無法按 session 刪除（舊數據）" 
      })).setMimeType(ContentService.MimeType.JSON);
    }

    const targetSid = String(sessionId).trim();
    let deleted = 0;

    // Delete from bottom to top (safe)
    for (let i = data.length - 1; i >= 1; i--) {
      const rowSid = String(data[i][checkCol] || "").trim();
      if (rowSid === targetSid) {
        sheet.deleteRow(i + 1);
        deleted++;
      }
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      deleted: deleted,
      session_id: targetSid
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ 
      status: "error", 
      message: err.toString() 
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * ========================================================================
 * NEW: replaceUserSession (strongly recommended for History major edits)
 * ========================================================================
 *
 * Completely replaces an entire training session (workout) while keeping the same session_id.
 *
 * Flow:
 *   1. Delete all existing rows with this sessionId (robust deleteUserSession logic)
 *   2. Insert the new exercises + sets using the **same sessionId**
 *
 * This is much cleaner than "delete + addWorkout" when the frontend wants to preserve
 * the original session_id (important for analytics, future references, etc.).
 *
 * Frontend usage (very simple):
 * {
 *   action: "replaceSession",
 *   user: "jeff",
 *   sessionId: "the-original-session-id",
 *   workout: {
 *     date: "2026-06-13",
 *     notes: "今天感覺不錯",
 *     exercises: [
 *       { name: "硬拉 (Deadlift)", sets: [ {weight: 80, reps: 5, notes: ""}, ... ] },
 *       ...
 *     ]
 *   }
 * }
 *
 * Returns: { status: "success", session_id, deleted: X, added: Y }
 */
function replaceUserSession(username, sessionId, workout) {
  try {
    if (!username || sessionId == null || String(sessionId).trim() === "") {
      return ContentService.createTextOutput(JSON.stringify({ 
        status: "error", 
        message: "缺少用戶或 sessionId" 
      })).setMimeType(ContentService.MimeType.JSON);
    }

    if (!workout || !Array.isArray(workout.exercises)) {
      return ContentService.createTextOutput(JSON.stringify({ 
        status: "error", 
        message: "workout.exercises 必須是陣列" 
      })).setMimeType(ContentService.MimeType.JSON);
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = _ensureUserLogSheet(ss, username);

    const targetSid = String(sessionId).trim();
    const dateStr = workout.date || '';
    const sessionNotes = workout.notes || workout.session_notes || '';

    // Step 1: Delete old rows for this session (reuse robust logic)
    let deleted = 0;
    const dataBefore = sheet.getDataRange().getValues();
    const headers = dataBefore[0].map(h => String(h).trim());
    const sidCol = headers.indexOf("session_id");

    if (sidCol >= 0) {
      for (let i = dataBefore.length - 1; i >= 1; i--) {
        if (String(dataBefore[i][sidCol] || "").trim() === targetSid) {
          sheet.deleteRow(i + 1);
          deleted++;
        }
      }
    }

    // Step 2: Insert new data using the same sessionId
    let added = 0;
    workout.exercises.forEach(ex => {
      (ex.sets || []).forEach(set => {
        const log = {
          id: set.id || set._clientLogId || (Date.now() + Math.random()),
          session_id: targetSid,
          date: dateStr,
          exercise: ex.name || '',
          reps: set.reps || 0,
          weight: set.weight || 0,
          body_weight: set.body_weight != null ? parseFloat(set.body_weight) : 0,
          duration: set.duration != null ? parseInt(set.duration) : 0,
          incline: set.incline != null ? parseFloat(set.incline) : 0,
          speed: set.speed != null ? parseFloat(set.speed) : 0,
          notes: set.notes || '',
          session_notes: sessionNotes,
          volume: set.volume
        };
        _appendLogRow(sheet, log);
        added++;
      });
    });

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      session_id: targetSid,
      deleted: deleted,
      added: added,
      message: `成功替換整個訓練記錄（刪除 ${deleted} 組，新增 ${added} 組）`
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ 
      status: "error", 
      message: "replaceSession 失敗: " + err.toString() 
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function saveUserWorkoutSet(username, setObj) {
  try {
    if (!username || !setObj || !setObj.name || !Array.isArray(setObj.exercises)) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Invalid set data" })).setMimeType(ContentService.MimeType.JSON);
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName("Workout_Sets");
    if (!sheet) {
      sheet = ss.insertSheet("Workout_Sets");
      sheet.appendRow(["user", "set_id", "name", "exercises_json", "updated"]);
    }
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["user", "set_id", "name", "exercises_json", "updated"]);
    }

    const data = sheet.getDataRange().getValues();
    const exercisesJson = JSON.stringify(setObj.exercises);
    const now = new Date();
    let id = setObj.id || `ws_${Date.now()}_${Math.floor(Math.random()*100000)}`;
    let found = false;

    if (setObj.id) {
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]).trim() === String(username).trim() &&
            String(data[i][1]).trim() === String(setObj.id).trim()) {
          sheet.getRange(i + 1, 3).setValue(setObj.name);
          sheet.getRange(i + 1, 4).setValue(exercisesJson);
          sheet.getRange(i + 1, 5).setValue(now);
          found = true;
          break;
        }
      }
    }

    if (!found) {
      sheet.appendRow([username, id, setObj.name, exercisesJson, now]);
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      set: { id: id, name: setObj.name, exercises: setObj.exercises }
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
